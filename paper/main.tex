% Unnumbered top-level headings in the CHAPTER style:
% INTRODUCTION, CONCLUSIONS, BIBLIOGRAPHY, APPENDICES...
\newcommand{\thesischapter}[1]{%
  \clearpage
  \begin{center}\bfseries\MakeUppercase{#1}\end{center}
  \vspace{0.5\baselineskip}
  % \phantomsection: without it every unnumbered chapter's TOC entry links
  % back to the start of the document instead of its own page.
  \phantomsection
  % \texorpdfstring: \MakeUppercase must not reach hyperref's PDF-string
  % expansion, where it breaks inside \csname ... \endcsname.
  \addcontentsline{toc}{section}{\texorpdfstring{\MakeUppercase{#1}}{#1}}%
  \markboth{#1}{#1}%
}

% \source{...} -- 10 pt line directly under a table/figure
\newcommand{\source}[1]{%
  \par\vspace{2pt}%
  \begingroup
    \fontsize{10}{12}\selectfont\setstretch{1}%
    \noindent\parbox{\linewidth}{%
      \raggedright\hangindent=1.4em\hangafter=1
      \textit{Source:} #1\par}%
  \endgroup\par\vspace{2pt}%
}

% ============================================================
%  8. EQUATIONS: centred, numbered [1], [2], ...
% ============================================================
% Applied at \begin{document} so it lands after microtype's own \tagform@
% patch -- redefining it in the preamble makes microtype warn and bail.
\AtBeginDocument{%
  \makeatletter
  \def\tagform@#1{\maketag@@@{[\ignorespaces#1\unskip\@@italiccorr]}}%
  \makeatother
}
